import sys, socket

#------------------------------------------------
def getIPandName(nomMachine):
    #nomMachine: اسم الجهاز الذي نريد الحصول على عنوانه IP
    # nomMachine-->عنوان IP
    try:
        ip=socket.gethostbyname(nomMachine)
        print "ip[%s]=%s" % (nomMachine,ip)
    except socket.error, erreur:
        print "ip[%s]=%s" % (nomMachine,erreur)
        return

    # العنوان IP --> nomMachine
    try:
        name=socket.gethostbyaddr(ip)
        print "name[%s]=%s" % (ip,name[0])
    except socket.error, erreur:
        print "name[%s]=%s" % (ip,erreur)
        return

import sys,socket

  #-----------------------------------------------------------------------
def getIndex(site):
    # يقرأ موقع URL ويخزنه في الملف site.html
  
    # في البداية لم تحدث أية أخطاء
    erreur=""
    # إنشاء الملف site.html
    try:
        html=open("%s.html" % (site),"w")
    except IOError, erreur:
        pass
    # خطأ؟
    if erreur:
        return "Erreur (%s) lors de la création du fichier %s.html" % (erreur,site)

    # فتح اتصال على المنفذ 80 للموقع
    try:
        connexion=socket.create_connection((site,80))
    except socket.error, erreur:
        pass
    # العودة في حالة حدوث خطأ
    if erreur:
        return "Echec de la connexion au site (%s,80) : %s" % (site,erreur)

    # يمثل الاتصال تدفقًا ثنائي الاتجاه للاتصالات
    # بين العميل (هذا البرنامج) وخادم الويب الذي تم الاتصال به
    # تُستخدم هذه القناة لتبادل الأوامر والمعلومات
    # بروتوكول الاتصال هو HTTP

    # يرسل العميل الأمر get لطلب URL /
    # صيغة الأمر get هي URL HTTP/1.0
    # يجب أن تنتهي رؤوس (headers) بروتوكول HTTP بسطر فارغ
    connexion.send("GET / HTTP/1.0\n\n")

    # سيقوم الخادم الآن بالرد على قناة الاتصال. سيقوم بإرسال جميع
    # بياناته ثم يغلق القناة. يقرأ العميل كل ما يصل من قناة الاتصال
    # حتى يتم إغلاق القناة
    ligne=connexion.recv(1000)
    while(ligne):
        html.write(ligne)
        ligne=connexion.recv(1000)

    # يقوم العميل بدوره بإغلاق الاتصال
    connexion.close()
    # إغلاق ملف html
    html.close()
    # العودة
    return "Transfert reussi du paragraphe index du site %s" % (site)

    11.3. عميل SMTP

    من بين بروتوكولات TCP / IP، يُعد SMTP (بروتوكول النقل SendMail) بروتوكول الاتصال الخاص بخدمة إرسال الرسائل.

    البرنامج (inet_03)
    
# -*- coding=utf-8 -*-

import sys,socket

#-----------------------------------------------------------------------
def getInfos(fichier):
    # يعرض المعلومات (SMTP، المرسل، المستلم، الرسالة) المأخوذة من الملف النصي [fichier]
    # السطر 1: SMTP، المرسل، المستلم
    # الأسطر التالية: نص الرسالة

    # فتح [fichier]
    erreur=""
    try:
        infos=open(fichier,"r")
    except IOError, erreur:
        return ("Le fichier %s n'a pu etre ouvert en lecture : %s" % (fichier, erreur))
  
    # قراءة السطر الأول
    ligne=infos.readline()
    # حذف علامة نهاية السطر
    ligne=cutNewLineChar(ligne)
    # استرداد الحقول smtp، المرسل، المستلم
    champs=ligne.split(",")
    # هل عدد الحقول صحيح؟
    if len(champs)!=3 :
        return ("La ligne 1 du fichier %s (serveur smtp, expediteur, destinataire) a un nombre de champs incorrect" % (fichier))
    # «معالجة» المعلومات المسترجعة
    # إزالة «المسافات» التي تسبق أو تتبع المعلومات المفيدة من كل حقل من الحقول الثلاثة
    for i in range(3):
        champs[i]=champs[i].strip()
    # استرجاع الحقول
    (smtpServer,expediteur,destinataire)=champs
    message=""
    # قراءة بقية الرسالة
    ligne=infos.readline()
    while ligne!='':
        message+=ligne
        ligne=infos.readline()
    infos.close()
    # العودة
    return ("",smtpServer,expediteur,destinataire,message)

#-----------------------------------------------------------------------
def sendmail(smtpServer,expediteur,destinataire,message,verbose):
    # إرسال الرسالة إلى خادم SMTP smtpserver نيابة عن المرسل
    # إلى المستلم. إذا كانت القيمة verbose=True، يتم تتبع التبادلات بين العميل والخادم

    # يتم استرداد اسم العميل
    try:
        client=socket.gethostbyaddr(socket.gethostbyname("localhost"))[0]
    except socket.error, erreur:
        return "Erreur IP / Nom du client : %s" % (erreur)
    
    # فتح اتصال على المنفذ 25 لـ smtpServer
    try:
        connexion=socket.create_connection((smtpServer,25))
    except socket.error, erreur:
        return "Echec de la connexion au site (%s,25) : %s" % (smtpServer,erreur)

    # يمثل الاتصال تدفقًا ثنائي الاتجاه للاتصالات
    # بين العميل (هذا البرنامج) وخادم SMTP الذي تم الاتصال به
    # يُستخدم هذا القناة لتبادل الأوامر والمعلومات

    # بعد الاتصال، يرسل الخادم رسالة ترحيب يتم قراءتها
    erreur=sendCommand(connexion,"",verbose,1)
    if(erreur) :
        connexion.close()
        return erreur
    # الأمر ehlo:
    erreur=sendCommand(connexion,"EHLO %s" % (client),verbose,1)
    if erreur :
        connexion.close()
        return erreur
    # الأمر mail from:
    erreur=sendCommand(connexion,"MAIL FROM: <%s>" % (expediteur),verbose,1)
    if erreur :
        connexion.close()
        return erreur
    # الأمر rcpt to:
    erreur=sendCommand(connexion,"RCPT TO: <%s>" % (destinataire),verbose,1)
    if erreur :
        connexion.close()
        return erreur
    # الأمر data
    erreur=sendCommand(connexion,"DATA",verbose,1)
    if erreur :
        connexion.close()
        return erreur
    # تحضير الرسالة المراد إرسالها
    # يجب أن تحتوي على الأسطر التالية
    # من: المرسل
    # إلى: المستلم
    # سطر فارغ
    # الرسالة
    # .
    data="From: %s\r\nTo: %s\r\n%s\r\n.\r\n" % (expediteur,destinataire,message)
    # إرسال الرسالة
    erreur=sendCommand(connexion,data,verbose,0)
    if erreur :
        connexion.close()
        return erreur
    # أمر الخروج
    erreur=sendCommand(connexion,"QUIT",verbose,1)
    if erreur :
        connexion.close()
        return erreur
    # نهاية
    connexion.close()
    return "Message envoye"

# --------------------------------------------------------------------------
def sendCommand(connexion,commande,verbose,withRCLF):
    # إرسال الأمر إلى قناة الاتصال
    # وضع التفصيل إذا كان verbose=1
    # إذا كان withRCLF=1، يُضاف التسلسل RCLF إلى الأمر

    # البيانات
    RCLF="\r\n" if withRCLF else ""
    # إرسال الأمر إذا كان الأمر غير فارغ
    if commande:
        connexion.send("%s%s" % (commande,RCLF))
        # رد محتمل
        if verbose:
            affiche(commande,1)
        # قراءة الرد الذي يقل عن 1000 حرف
        reponse=connexion.recv(1000)
        # رد محتمل
        if verbose:
            affiche(reponse,2)
        # استرداد رمز الخطأ
        codeErreur=reponse[0:3]
        # هل أرسل الخادم خطأً؟
        if int(codeErreur) >=500:
            return reponse[4:]
        # العودة دون أخطاء
        return ""

# --------------------------------------------------------------------------
def affiche(echange,sens):
    # عرض التبادل؟ على الشاشة
    # إذا كانت القيمة sens=1، يعرض -->التبادل
    # إذا كانت القيمة sens=2، يُعرض <-- التبادل بدون الحرفين الأخيرين RCLF
    if sens==1:
        print "--> [%s]" % (echange)
        return
    elif sens==2:
        l=len(echange)
        print "<-- [%s]" % echange[0:l-2]
        return

      
# --------------------------------------------------------------------------
def cutNewLineChar(ligne):
    # يتم حذف علامة نهاية السطر من [ligne] إن وجدت
    l=len(ligne)
    while(ligne[l-1]=="\n" or ligne[l-1]=="\r"):
        l-=1
    return(ligne[0:l])

    11.6. خادم TCP عام

    نقترح كتابة برنامج نصي بلغة Python

    • سيكون بمثابة خادم TCP قادر على خدمة عميل واحد في كل مرة،
    • يقبل أسطر النص المرسلة من العميل،
    • ويقبل سطور النص الواردة من لوحة المفاتيح والتي يتم إرسالها كرد على العميل.

    وبالتالي، فإن المستخدم الذي يستخدم لوحة المفاتيح هو الذي يقوم بدور الخادم:

    • حيث يرى على شاشته الأسطر النصية المرسلة من العميل؛
    • ويرد عليه عن طريق كتابة الرد على لوحة المفاتيح.

    وبذلك يمكنه التكيف مع أي نوع من العملاء. ولهذا السبب سنطلق عليه اسم «خادم TCP عام». وهو أداة مفيدة لاكتشاف بروتوكولات الاتصال TCP. في المثال التالي، سيكون عميل TCP هو متصفح ويب، مما سيسمح لنا باكتشاف بروتوكول HTTP الذي يستخدمه عملاء الويب.

    • السطر 14: سيتواصل خادم TCP مع العميل باستخدام خيطين

      • خيط لكتابة الأسطر إلى العميل.
    • السطر 14: يتم إنشاء مؤشر الترابط الخاص بقراءة الأسطر الواردة من العميل. ويحدد المعامل الخاص به target الطريقة التي ينفذها مؤشر الترابط. وهي الطريقة المحددة في السطر 25؛
    • السطر 25: يتم تشغيل مؤشر الترابط الخاص بالقراءة؛
    • الأسطر 25-32: الطريقة التي ينفذها مؤشر الترابط الخاص بالقراءة؛
    • السطر 28: يقوم مؤشر الترابط الخاص بالقراءة بقراءة جميع أسطر النص المرسلة من العميل حتى استلام السطر "bye"؛
    • السطر 18: نحن هنا في مؤشر الترابط الخاص بالكتابة إلى العميل. المبدأ هو إرسال جميع أسطر النص التي يكتبها المستخدم على لوحة المفاتيح إلى العميل.
