    9.2. تثبيت MySQL

    • خادم ويب Apache. سنستخدمه لكتابة نصوص برمجية ويب بلغة Python؛
    • برنامج SGBD و MySQL؛
    • لغة البرمجة النصية PHP؛
    • أداة لإدارة SGBD MySQL مكتوبة بلغة PHP: phpMyAdmin.

    يمكن تنزيل WampServer (يونيو 2011) من العنوان التالي:

    9.5. إنشاء جدول MySQL

    الآن بعد أن تعلمنا كيفية إنشاء اتصال باستخدام SGBD MySQL، سنبدأ في إصدار الأوامر SQL عبر هذا الاتصال. للقيام بذلك، سنقوم بالاتصال بقاعدة البيانات التي تم إنشاؤها ([dbpersonnes]) واستخدام هذا الاتصال لإنشاء جدول داخل قاعدة البيانات.

    البرنامج (mysqldb_03)
    
# استيراد الوحدة النمطية MySQLdb
import sys
sys.path.append("D:\Programs\ActivePython\site-packages")
import MySQLdb

# ---------------------------------------------------------------------------------
def executeSQL(connexion,update):
    # تنفيذ طلب تحديث على الاتصال
    # طلب مؤشر
    curseur=connexion.cursor()
    # تنفيذ استعلام SQL على الاتصال
    try:
        curseur.execute(update)
        connexion.commit()
    except Exception, erreur:
        connexion.rollback()
        raise
    finally:
        curseur.close()

# ---------------------------------------------- main
# الاتصال بقاعدة البيانات MySQL
# هوية المستخدم
ID="admpersonnes"
PWD="nobody"
# الجهاز المضيف لنظام إدارة قواعد البيانات
HOTE="localhost"
# هوية قاعدة البيانات
BASE="dbpersonnes"
# الاتصال
try:
    connexion=MySQLdb.connect(host=HOTE,user=ID,passwd=PWD,db=BASE)
except MySQLdb.OperationalError,message:
    print message
    sys.exit()

# حذف جدول الأشخاص إن وجد
# وإذا لم تكن موجودة، فسيحدث خطأ
# يتم تجاهلها
requete="drop table personnes"

    executeSQL(connexion,requete)
except:
    pass
# إنشاء جدول الأشخاص
requete="create table personnes (prenom varchar(30) NOT NULL, nom varchar(30) NOT NULL, age integer NOT NULL, primary key(nom,prenom))"

    executeSQL(connexion,requete)
except MySQLdb.OperationalError,message:
    print message
    sys.exit()
# يتم قطع الاتصال والخروج
try:
    connexion.close()

    print message

    بعد أن أنشأنا سابقًا الجدول [personnes]، سنقوم الآن بملئه.

# استيراد الوحدة النمطية MySQLdb
import sys
sys.path.append("D:\Programs\ActivePython\site-packages")
import MySQLdb
# وحدات أخرى
import re

# ---------------------------------------------------------------------------------
def executerCommandes(HOTE,ID,PWD,BASE,SQL,suivi=False,arret=True):
    # استخدام الاتصال (HOTE، ID، PWD، BASE)
    # يُنفذ الأوامر SQL الموجودة في الملف النصي SQL عبر هذا الاتصال
    # هذا الملف هو ملف أوامر SQL يتم تنفيذها بمعدل أمر واحد في كل سطر
    # إذا كانت القيمة «متابعة» = «صحيح»، فإن كل عملية تنفيذ لأمر من أوامر SQL تُعرض مع إشارة إلى نجاحها أو فشلها
    # إذا كانت arret=True، تتوقف الدالة عند أول خطأ تصادفه، وإلا فإنها تنفذ جميع أوامر SQL
    # تُرجع الدالة قائمة (عدد الأخطاء، الخطأ 1، الخطأ 2، ...)

    # يتم التحقق من وجود الملف SQL
    data=None
    try:
        data=open(SQL,"r")
    except:
        return [1,"Le fichier %s n'existe pas" % (SQL)]

    # الاتصال
    try:
        connexion=MySQLdb.connect(host=HOTE,user=ID,passwd=PWD,db=BASE)
    except MySQLdb.OperationalError,erreur:
        return [1,"Erreur lors de la connexion a MySQL sous l'identite (%s,%s,%s,%s) : %s" % (HOTE, ID, PWD, BASE, erreur)]
    
    # يتم طلب مؤشر
    curseur=connexion.cursor()
    # تنفيذ الاستعلامات SQL الموجودة في الملف SQL
    # يتم وضعها في مصفوفة
    requetes=data.readlines()
    # يتم تنفيذها واحدة تلو الأخرى - في البداية لم تحدث أخطاء
    erreurs=[0]
    for i in range(len(requetes)):
        # يتم حفظ الاستعلام الحالي
        requete=requetes[i]
        # هل الاستعلام فارغ؟ إذا كان الجواب نعم، ننتقل إلى الاستعلام التالي
        if re.match(r"^\s*$",requete):
            continue
        # تنفيذ الاستعلام i
        erreur=""
        try:
            curseur.execute(requete)
        except Exception, erreur:
            pass
        #هل حدث خطأ؟
        if erreur:
            # خطأ آخر
            erreurs[0]+=1
            # رسالة خطأ
            msg="%s : Erreur (%s)" % (requete,erreur)
            erreurs.append(msg)
            # هل يتم عرض الشاشة أم لا؟
            if suivi:
                print msg
            # هل نتوقف؟
            if arret:
                return erreurs
        else:
            if suivi: 
                print "%s : Execution reussie" % (requete)
    # إغلاق الاتصال وتحرير الموارد
    curseur.close()
    connexion.commit()
    # يتم قطع الاتصال
    try:
        connexion.close()
    except MySQLdb.OperationalError,erreur:
        # خطأ آخر
        erreurs[0]+=1
        # رسالة خطأ
        msg="%s : Erreur (%s)" % (requete,erreur)
        erreurs.append(msg)

    # العودة
    return erreurs


# ---------------------------------------------- main

#عرض عدد الأخطاء

    9.7. تنفيذ أي استعلامات SQL

    يتيح البرنامج النصي التالي تنفيذ ملف أوامر SQL وعرض نتيجة كل منها:

    • نتيجة الأمر SELECT إذا كان الأمر هو SELECT؛
    • عدد الأسطر التي تم تعديلها إذا كان الأمر هو INSERT أو UPDATE أو DELETE.
    البرنامج (mysqldb_05)

def cutNewLineChar(ligne):
    # حذف علامة نهاية السطر من [ligne] إن وجدت
    l=len(ligne)
    while(ligne[l-1]=="\n" or ligne[l-1]=="\r"):
        l-=1

# ---------------------------------------------------------------------------------
def afficherInfos(curseur):
    # عرض نتيجة استعلام SQL
    # هل كان هذا استعلام SELECT؟
    if curseur.description:
        # يوجد وصف — لذا فهو أمر SELECT
        # description[i] هو وصف العمود رقم i في أمر select
        # descriptionQZXW2HTMLBW2ldZQXQZXW2HTMLBWzBdZQX هو اسم العمود رقم i في قائمة الاختيار
        # يتم عرض أسماء الحقول
        titre=""
        for i in range(len(curseur.description)):
            titre+=curseur.description[i][0]+","
        # يتم عرض قائمة الحقول بدون الفاصلة الختامية
        print titre[0:len(titre)-1]
        # خط فاصل
        print "-"*(len(titre)-1)
        # السطر الحالي في قائمة الاختيار
        ligne=curseur.fetchone()
        while ligne:
            print ligne
            # السطر التالي في قائمة الاختيار
            ligne=curseur.fetchone()
    else:
        # لا توجد حقول - لم يكن هذا قائمة
        print "%s lignes(s) a (ont) ete modifiee(s)" % (curseur.rowcount)


# ---------------------------------------------------------------------------------
def executerCommandes(HOTE,ID,PWD,BASE,SQL,suivi=False,arret=True):
    # يستخدم الاتصال (HOTE،ID،PWD،BASE)
    # يُنفذ عبر هذا الاتصال الأوامر SQL الموجودة في الملف النصي SQL
    # هذا الملف هو ملف أوامر SQL يتم تنفيذها بمعدل أمر واحد في كل سطر
    # إذا كانت القيمة «متابعة» تساوي 1، فإن كل عملية تنفيذ لأمر من أوامر SQL تُعرض مع إشارة إلى نجاحها أو فشلها
    # إذا كانت القيمة «arret» تساوي 1، تتوقف الدالة عند أول خطأ تصادفه، وإلا فإنها تنفذ جميع أوامر SQL
    # تُرجع الدالة مصفوفة (عدد الأخطاء، الخطأ 1، الخطأ 2، ...)

    # يتم التحقق من وجود الملف SQL
    data=None
    try:
        data=open(SQL,"r")
    except:
        return [1,"Le fichier %s n'existe pas" % (SQL)]

    # تسجيل الدخول
    try:
        connexion=MySQLdb.connect(host=HOTE,user=ID,passwd=PWD,db=BASE)
    except MySQLdb.OperationalError,erreur:
        return [1,"Erreur lors de la connexion a MySQL sous l'identite (%s,%s,%s,%s) : %s" % (HOTE, ID, PWD, BASE, erreur)]
    
    # يتم طلب مؤشر
    curseur=connexion.cursor()
    # تنفيذ الاستعلامات SQL الموجودة في الملف SQL
    # يتم وضعها في مصفوفة
    requetes=data.readlines()
    # يتم تنفيذها واحدة تلو الأخرى - في البداية لم تحدث أخطاء
    erreurs=[0]
    for i in range(len(requetes)):
        # يتم حفظ الاستعلام الحالي
        requete=requetes[i]
        # هل الاستعلام فارغ؟ إذا كان الجواب نعم، ننتقل إلى الاستعلام التالي
        if re.match(r"^\s*$",requete):
            continue
        # تنفيذ الاستعلام i
        erreur=""
        try:
            curseur.execute(requete)
        except Exception, erreur:
            pass
        #هل حدث خطأ؟
        if erreur:
            # خطأ آخر
            erreurs[0]+=1
            # رسالة خطأ
            msg="%s : Erreur (%s)" % (requete,erreur)
            erreurs.append(msg)
            # هل يتم عرض الشاشة أم لا؟
            if suivi:
                print msg
            # هل نتوقف؟
            if arret:
                return erreurs
        else:
            if suivi: 
                print "%s : Execution reussie" % (requete)
                # معلومات حول نتيجة الاستعلام الذي تم تنفيذه
                afficherInfos(curseur)

    # إغلاق الاتصال وتحرير الموارد
    curseur.close()
    connexion.commit()
    # يتم قطع الاتصال
    try:
        connexion.close()
    except MySQLdb.OperationalError,erreur:
        # خطأ آخر
        erreurs[0]+=1
        # رسالة خطأ
        msg="%s : Erreur (%s)" % (requete,erreur)
        erreurs.append(msg)

    # العودة
    return erreurs
